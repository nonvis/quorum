# macOS: Homebrew's OpenSSL is keg-only, so CMake's FindOpenSSL can't locate it
# without a hint. Query Homebrew for the prefix when OPENSSL_ROOT_DIR isn't set.
if(APPLE AND NOT DEFINED OPENSSL_ROOT_DIR)
    find_program(BREW_EXECUTABLE brew)
    if(BREW_EXECUTABLE)
        execute_process(
            COMMAND ${BREW_EXECUTABLE} --prefix openssl@3
            OUTPUT_VARIABLE OPENSSL_ROOT_DIR
            OUTPUT_STRIP_TRAILING_WHITESPACE
            ERROR_QUIET
        )
    endif()
endif()

# ── Build identity (generated header) ──
# quorum_version.h carries the git short sha, the dirty flag, the stamp time and
# the project VERSION. It is regenerated by an ALWAYS-RUN build target, not only
# at configure time -- a configure-time-only stamp keeps printing the sha of the
# tree that was checked out when cmake last ran. The script rewrites the header
# only when the identity actually changed, so an unchanged tree never forces a
# relink of main.cpp.
set(QUORUM_VERSION_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(QUORUM_VERSION_HEADER ${QUORUM_VERSION_DIR}/quorum_version.h)
set(QUORUM_VERSION_STAMP ${CMAKE_CURRENT_SOURCE_DIR}/src/version_stamp.cmake)
set(QUORUM_VERSION_STAMP_ARGS
    -DQ_SRC_DIR=${CMAKE_CURRENT_SOURCE_DIR}/..
    -DQ_OUT_FILE=${QUORUM_VERSION_HEADER}
    -DQ_VERSION=${PROJECT_VERSION}
)
file(MAKE_DIRECTORY ${QUORUM_VERSION_DIR})
# Seed once at configure time so the header exists for the first dependency scan.
execute_process(COMMAND ${CMAKE_COMMAND} ${QUORUM_VERSION_STAMP_ARGS}
                        -P ${QUORUM_VERSION_STAMP})
add_custom_target(quorum_version_stamp ALL
    COMMAND ${CMAKE_COMMAND} ${QUORUM_VERSION_STAMP_ARGS} -P ${QUORUM_VERSION_STAMP}
    BYPRODUCTS ${QUORUM_VERSION_HEADER}
    COMMENT "Stamping build identity into quorum_version.h"
    VERBATIM
)

# Test I reads templates/skills/quorum-roles/evaluator/SKILL.md. Its cwd ladder
# ("templates", "../templates", "../../templates") only resolves when ctest runs
# from inside the repo, so the test went RED from any out-of-tree build dir.
# QUORUM_SOURCE_ROOT pins the repo root at COMPILE time.
# Chosen over a WORKING_DIRECTORY on add_test(): tests A-G chdir into their own
# make_temp_dir() and restore to whatever cwd they started in, so moving the
# start cwd would change the surface those tests exercise. A compile definition
# leaves their cwd behaviour byte-for-byte unchanged.
target_compile_definitions(test_agent_create PRIVATE
    QUORUM_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME test_agent_create COMMAND test_agent_create)
